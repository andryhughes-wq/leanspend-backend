const adCache = new Map();
const CACHE_TTL = 6 * 60 * 60 * 1000;

function isCacheValid(key) {
  const entry = adCache.get(key);
  return entry && (Date.now() - entry.timestamp < CACHE_TTL);
}

function clearAdCache() {
  adCache.clear();
  logger.info('Ad cache cleared');
}

const HEADERS = {
  'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/124.0.0.0 Safari/537.36',
  'Accept': 'application/json, text/html, */*',
  'Accept-Language': 'en-US,en;q=0.9',
};

// Try to fetch HEB deals from their public API
async function fetchHEBDeals() {
  try {
    const resp = await axios.get('https://www.heb.com/pharmacy-api/v1/weekly-deals', {
      headers: HEADERS, timeout: 10000
    });
    if (resp.data && resp.data.deals) {
      return resp.data.deals.slice(0, 30).map(d => ({
        product_name: d.name || d.title,
        deal_price: d.salePrice || d.price,
        original_price: d.regularPrice,
        savings_amount: d.regularPrice && d.salePrice ? (d.regularPrice - d.salePrice).toFixed(2) : null,
        category: d.category || 'Grocery',
        store: 'heb', store_name: 'HEB', store_slug: 'heb',
        valid_from: new Date().toISOString().split('T')[0],
        valid_to: new Date(Date.now() + 7*24*60*60*1000).toISOString().split('T')[0],
        live: true, source: 'heb_api'
      }));
    }
  } catch (err) {
    logger.warn('HEB API failed, using static:', err.message);
  }
  return null;
}

// Try Walmart open endpoint
async function fetchWalmartDeals() {
  try {
    const resp = await axios.get('https://www.walmart.com/cp/grocery/976759?moduleType=category&moduleId=category-page&facet=shelf_description%3AGrocery', {
      headers: { ...HEADERS, 'Accept': 'application/json' },
      timeout: 10000
    });
    // Walmart blocks most scraping - fall through to static
  } catch (err) {
    logger.warn('Walmart scrape failed, using static:', err.message);
  }
  return null;
}

function getWeekNumber() {
  const now = new Date();
  const start = new Date(now.getFullYear(), 0, 1);
  return Math.floor((now - start) / (7 * 24 * 60 * 60 * 1000));
}

async function scrapeAllWeeklyAds(storeNames = ['kroger','walmart','heb','target','aldi','costco','samsclub','safeway','randalls']) {
  const cacheKey = storeNames.join(',');
  if (isCacheValid(cacheKey)) {
    logger.info('Returning cached deals');
    return adCache.get(cacheKey).data;
  }

  let allDeals = [];
  const week = getWeekNumber();

  for (const store of storeNames) {
    try {
      let storeDeals = [];

      if (['kroger','krogers','fred meyer','ralphs','king soopers'].includes(store.toLowerCase())) {
        // Live Kroger API
        try {
          const krogerDeals = await krogerService.getWeeklyDeals();
          if (krogerDeals && krogerDeals.length > 0) {
            storeDeals = krogerDeals.map(d => ({
              ...d,
              store: 'kroger', store_name: 'Kroger', store_slug: 'kroger',
              live: true, source: 'kroger_api'
            }));
            logger.info(`Kroger API: ${storeDeals.length} deals`);
          }
        } catch (err) {
          logger.warn('Kroger API failed:', err.message);
          storeDeals = getRotatingDeals('kroger', week);
        }
      } else if (store.toLowerCase() === 'heb') {
        // Try live HEB first
        const hebDeals = await fetchHEBDeals();
        storeDeals = hebDeals || getRotatingDeals('heb', week);
        logger.info(`HEB: ${storeDeals.length} deals (${hebDeals ? 'live' : 'static'})`);
      } else {
        // Rotating static for all other stores
        storeDeals = getRotatingDeals(store, week);
        logger.info(`${store}: ${storeDeals.length} static deals (week ${week})`);
      }

      allDeals = [...allDeals, ...storeDeals];
    } catch (err) {
      logger.error(`Error fetching ${store}:`, err.message);
    }
  }

  const result = {
    deals: allDeals,
    storeCount: storeNames.length,
    totalDeals: allDeals.length,
    fetchedAt: new Date().toISOString(),
    week: week,
  };

  adCache.set(cacheKey, { data: result, timestamp: Date.now() });
  logger.info(`Total deals: ${allDeals.length} across ${storeNames.length} stores`);
  return result;
}
